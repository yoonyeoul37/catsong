/**
 * CPBC 구동 클래스
 * 외부 리소스 호출 및 페이지 라우팅
 * @class
 */
class Cpbc extends HTMLElement {
  constructor() {
    super();

    this.view = {
      env: '',        // local, production
      javascript: {
        local: '/javascript',
        production: '/javascript'
      },
      style: {
        local: '/style',
        production: '/style'
      }
    };
  }

  static get observedAttributes() {
    return Cpbc.getProperty().map(property => property.name);
  }

  static getProperty(self) {
    return [{
      name: 'env',
      bind: env => {
        self.view.env = env;
      }
    }];
  }

  async connectedCallback() {
    try {
      // window._babelPolyfill = false;  // 바벨 의존성 체크(HTML5 플레이어)
      await this.loadResource();

      CPBC_GLOBAL.ENV.IS_MOBILE = (parsed => {
        if(parsed.type === 'Desktop' && ['Mac', 'Linux'].includes(parsed.os)) {
          return navigator.maxTouchPoints > 0;
        }
        else {
          return ['Mobile', 'Tablet'].includes(parsed.type);
        }
      })(DeviceDetector.parse(platform.ua));

      if(CPBC_GLOBAL.ENV.IS_MOBILE) {
        location.href = `https://m.cpbc.co.kr${location.pathname}${location.search}`;

        return;
      }

      CpbcBase.import(this);
      CpbcPath.import(this);
      CpbcEvent.import(this);

      this.render();
      this.renderContent(this.getPage());

      document.querySelector('#skip_nav').style = '';           // 스킵네비게이션 활성화

      window.dispatchEvent(new Event('resize'));
    }
    catch(error) {
      console.error(error);
    }
  }

  async attributeChangedCallback(name, oldValue, newValue) {
    try {
      Cpbc.getProperty(this).find(property => property.name === name).bind(newValue);

      if(this.innerHTML.length) {
        await this.loadResource();

        CpbcBase.import(this);
        CpbcPath.import(this);
        CpbcEvent.import(this);

        this.render();
        this.renderContent(this.getPage());

        window.dispatchEvent(new Event('resize'));
      }
    }
    catch(error) {
      console.error(error);
    }
  }

  render() {
    try {
      this.innerHTML = `<div id="container" class="container"></div>`;
    }
    catch(error) {
      this.log(error);
    }
  }

  /**
   * 페이지 라우팅
   * @param page 컴포넌트 이름
   */
  renderContent(page) {
    try {
      this.querySelector('#container').innerHTML = `<cpbc-${page}></cpbc-${page}>`;
    }
    catch(error) {
      this.log(error);
    }
  }

  /**
   * 외부 리소스(스크립트) DOM 주입
   * @async
   * @param path 리소스 경로
   * @return {Promise<unknown>}
   */
  async appendScript(path) {
    return new Promise((resolve, reject) => {
      const $script = document.createElement('script');

      $script.type = 'text/javascript';
      $script.src = path;
      $script.async = false;
      $script.onload = () => {
        resolve();
      };
      $script.onerror = () => {
        reject('Failed to load script resource.');
      };

      document.querySelector('head').appendChild($script);
    });
  };

  /**
   * 외부 및 내부 리소스 로드
   * @async
   * @see https://www.npmjs.com/package/loadjs
   * @return {Promise<void>}
   */
  async loadResource() {
    try {
      const { env, javascript, style } = this.view;
      const path = {
        javascript: javascript[env],
        style: style[env]
      };

      await this.appendScript(`${javascript[env]}/loadjs.min.js`);
      
      // 외부 리소스 먼저 로드
      await loadjs([
        `${path.javascript}/jquery-1.12.4.min.js`,
        `${path.javascript}/platform-1.3.6.js`,
        `${path.javascript}/devicedetector-1.0.2.min.js`,
        `${path.javascript}/underscore-1.13.1.min.js`,
        `${path.javascript}/swiper-8.3.2.min.js`,
        `//static.cpbc.co.kr/gnb/javascript/cpbc-gnb-desktop-1.0.0.js?t=${new Date().getTime()}`,
        '//static.cpbc.co.kr/player/javascript/cpbc-player-1.0.0.js',
        `${path.style}/cpbc-home-desktop-1.0.0.min.css`,
        '//static.cpbc.co.kr/gnb/style/cpbc-gnb-desktop-1.0.0.min.css',
        '//static.cpbc.co.kr/player/style/cpbc-player-1.0.0.min.css'
      ], {
        async: true,
        returnPromise: true
      });

      // 외부 리소스 로드 후 내부 리소스 로드
      await loadjs([
        `${path.javascript}/cpbc-home-desktop-1.0.0${{production: '.min', local: ''}[env]}.js`,
        '//board.cpbc.co.kr/pc/js/sbs_board_module_nodomain.js',
        '//board.cpbc.co.kr/pc/css/board-1.0.0.min.css'
      ], {
        before: (path, element) => {
          if (path.includes('sbs_board_module')) {
            element.setAttribute('id', 'SBSBoardModule');
          }
        },
        returnPromise: true
      });
    }
    catch(error) {
      console.log(error);
    }
  }

  /**
   * 현재 페이지 정보 반환
   * @see CpbcPath
   * @return {string}
   */
  getPage() {
    try {
      const page = this.path.get().page;

      return page.length > 0 ? page : 'main';
    }
    catch(error) {
      this.log(error);
    }
  }

  /**
   * 디버깅용 핸들러
   * @see CpbcEvent
   * @param event
   */
  resized(event) {
    /*this.log({
      clientWidth: document.documentElement.clientWidth,
      clientHeight: document.documentElement.clientHeight,
      activeElement: document.activeElement
    });*/

    // this.log(event);
  }
}

customElements.define('cpbc-app', Cpbc);
